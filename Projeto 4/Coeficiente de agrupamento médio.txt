    /*Coeficiente de agrupamento medio*/

    for(i=0;i<N;i++)
    {
        for(j=0;j<N;j++)
        {
            if(W[i][j]!=0)
            {
                W[i][j] = 1;
            }
            else
            {
                W[i][j] = 0;
            }
        }
    }

    /*Construção da matriz binaria*/
    float B[N][N];
    for(i=0;i<N;i++)
    {
        for(j=0;j<N;j++)
        {
            W[i][j] = powf(W[i][j],(1.0/3.0));
            if(W[i][j] != 0)
            {
                B[i][j] = 1;
            }
            else
            {
                B[i][j] = 0;
            }
        }
    }
    float Q[N][N];
    for(i=0;i<N;i++)
    {
        for(k=0;k<N;k++)
        {
            Q[i][k] = 0;
            for(j=0;j<N;j++)
            {
                Q[i][k] = Q[i][k] + B[i][j] + B[j][k];
            }
        }
    }

    float Dtotal[N],Di[N];
    for(i=0;i<N;i++)
    {
        Dtotal[i] = 0;
        for(j=0;j<N;j++)
        {
            Dtotal[i] = Dtotal[i] + B[j][i] + B[i][j];
        }
    }
    /*Fazendo o vetor Di*/
    for(i=0;i<N;i++)
    {
        Di[i] = Q[i][i];
    }

    float A[N][N],C[N][N],D[N][N];
    for(i=0;i<N;i++)
    {
        for(j=0;j<N;j++)
        {
            A[i][j] = W[i][j] + W[j][i];
        }
    }
    /*Usando a matriz C*/
    for(i=0;i<N;i++)
    {
        for(k=0;k<N;k++)
        {
            C[i][k] = 0;
            for(j=0;j<N;j++)
            {
                C[i][k] = C[i][k] + A[i][j]*A[j][k];
            }
        }
    }
    /*Usando a matriz D*/
    for(i=0;i<N;i++)
    {
        for(k=0;k<N;k++)
        {
            D[i][k] = 0;
            for(j=0;j<N;j++)
            {
                D[i][k] = D[i][k] + C[i][j]*A[j][k];
            }
        }
    }
    printf("Coeficiente de agrupamento:\n");
    float coef[N], soma_coef = 0, coeficiente;
    for(i=0;i<N;i++)
    {
        coef[i] = B[i][i]/(2.0*(Dtotal[i]*(Dtotal[i]-1.0)-2.0*Di[i]));
        soma_coef = soma_coef + coef[i];
    }
    coeficiente = soma_coef/N;
    printf("\nCoeficiente de agrupamento medio: %f\n",coeficiente);


    /*Coeficiente de agrupamento médio - Sem Peso*/
    float coef_total = 0, coef, coef_global;
    int numero_triangulos = 0, vertice_grau = 0;
    for(i=0;i<N;i++)
    {
        for(j=0;j<N;j++)
        {
            if(MA[i][j] != 0)
            {
                vertice_grau = vertice_grau + 1;
                for(k=j+1;k<N;k++)
                {
                    if(MA[i][k] != 0 && MA[j][k] != 0)
                    {
                        numero_triangulos = numero_triangulos + 1;
                    }
                }
            }
        }
        if(vertice_grau > 1)
        {
            coef = (2*numero_triangulos)/(vertice_grau*(vertice_grau-1));
            coef_total = coef_total + coef;
        }
    }
    coef_global = coef_total/N;
    printf("Coeficiente de agrupamento Global:%f\n");
